🔌 API & Python SDK

Python SDK — Quick Start
pip install robo-safe

from robo_safe import SafetyEngine, Action

engine = SafetyEngine()
action = Action(name="move_arm", params={"speed": 1.5})
result = engine.validate(action)

print(result.status)  # ExecutionStatus.APPROVED
print(result.blocked) # False
